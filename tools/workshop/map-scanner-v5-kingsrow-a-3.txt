rule("Scanner: setup")
{
	event
	{
		Ongoing - Global;
	}

	actions
	{
		Disable Built-In Game Mode Completion;
		Disable Inspector Recording;
	}
}

rule("Scanner: pause match time")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Is Game In Progress == True;
	}

	actions
	{
		Pause Match Time;
	}
}

rule("Scanner: player HUD")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	actions
	{
		Event Player.Done = False;
		Event Player.Scanning = False;
		Event Player.J = 0;
		Event Player.NZ = 176;
		Create HUD Text(Event Player, Custom String("OW-TACTICS MAP SCANNER v5 - KINGS ROW A - CHUNK 3 / 3"), Custom String("Crouch + F: start scan"), Null, Left, 0, Color(Yellow), Color(White), Color(White), Visible To and String, Default Visibility);
		Create HUD Text(Event Player, Null, Null, Custom String("Pos {0}", Position Of(Event Player)), Left, 1, Color(White), Color(White), Color(White), Visible To and String, Default Visibility);
		Create HUD Text(Event Player, Null, Null, Custom String("Row {0} / {1}   Scanning {2}", Event Player.J, Event Player.NZ, Event Player.Scanning), Left, 2, Color(White), Color(White), Color(White), Visible To and String, Default Visibility);
		Create HUD Text(Event Player, Null, Null, Custom String("Done {0}", Event Player.Done), Left, 3, Color(White), Color(White), Color(White), Visible To and String, Default Visibility);
	}
}

rule("Scanner: scan chunk 3")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is Button Held(Event Player, Button(Interact)) == True;
		Is Button Held(Event Player, Button(Crouch)) == True;
		Event Player.Scanning == False;
		Event Player.Done == False;
	}

	actions
	{
		Event Player.Scanning = True;
		Event Player.Step = 0.5;
		Event Player.X0 = -5;
		Event Player.NX = 66;
		Event Player.Z0 = -62;
		Event Player.NZ = 176;
		Event Player.YTop = 90;
		Event Player.YBot = -40;
		Event Player.Layers = 6;
		Small Message(Event Player, Custom String("Chunk 3 started"));
		Enable Inspector Recording;
		Global.Row = Array(-1, Event Player.X0, Event Player.Z0, Event Player.Step, Event Player.NX, Event Player.NZ, Event Player.YTop, Event Player.YBot, Event Player.Layers, 3, 2);
		Disable Inspector Recording;
		For Player Variable(Event Player, J, 0, Event Player.NZ + 1, 1);
			Event Player.Temp = Array(Event Player.J);
			Event Player.Pz = Event Player.Z0 + Event Player.J * Event Player.Step;
			For Player Variable(Event Player, I, 0, Event Player.NX + 1, 1);
				Event Player.Px = Event Player.X0 + Event Player.I * Event Player.Step;
				Event Player.YFrom = Event Player.YTop;
				For Player Variable(Event Player, K, 0, Event Player.Layers, 1);
					Event Player.HitPos = Ray Cast Hit Position(Vector(Event Player.Px, Event Player.YFrom, Event Player.Pz), Vector(Event Player.Px, Event Player.YBot, Event Player.Pz), Null, All Players(All Teams), False);
					If(Y Component Of(Event Player.HitPos) > Event Player.YBot + 0.05);
						Event Player.Code = (Round To Integer(Y Component Of(Event Player.HitPos) * 20, To Nearest) + 1000) * 100 + Round To Integer(Max(0, Y Component Of(Ray Cast Hit Normal(Vector(Event Player.Px, Event Player.YFrom, Event Player.Pz), Vector(Event Player.Px, Event Player.YBot, Event Player.Pz), Null, All Players(All Teams), False))) * 99, To Nearest);
						Modify Player Variable(Event Player, Temp, Append To Array, Event Player.Code);
						Event Player.YFrom = Y Component Of(Event Player.HitPos) - 0.3;
					Else;
						Modify Player Variable(Event Player, Temp, Append To Array, 0);
						Event Player.YFrom = Event Player.YBot;
					End;
				End;
				Wait(0.016, Ignore Condition);
			End;
			Enable Inspector Recording;
			Global.Row = Event Player.Temp;
			Disable Inspector Recording;
		End;
		Enable Inspector Recording;
		Global.Row = Array(-2, 3);
		Disable Inspector Recording;
		Event Player.Done = True;
		Event Player.Scanning = False;
		Big Message(Event Player, Custom String("Chunk 3 done - copy log (Global, CSV all)"));
	}
}
